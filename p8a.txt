  <script>
    const tabs = ['home', 'closet', 'sales', 'wants', 'outfits', 'pinterest'];
    function switchTab(tId) {
      tabs.forEach(t => {
        const v = document.getElementById('view-' + t); if (v) v.classList.add('hidden');
        const b = document.getElementById('tb-' + t);
        if (b) { b.classList.remove('bg-cyan-400', 'text-slate-950', 'font-bold'); b.classList.add('glass', 'text-slate-300'); }
      });
      const av = document.getElementById('view-' + tId); if (av) av.classList.remove('hidden');
      const ab = document.getElementById('tb-' + tId);
      if (ab) { ab.classList.remove('glass', 'text-slate-300'); ab.classList.add('bg-cyan-400', 'text-slate-950', 'font-bold'); }
    }
    function openModal() { document.getElementById('cam-modal').classList.remove('hidden'); }
    function closeModal() { document.getElementById('cam-modal').classList.add('hidden'); }
    function openDrawerModal() { document.getElementById('drawer-select-modal').classList.remove('hidden'); }
    function closeDrawerModal() { document.getElementById('drawer-select-modal').classList.add('hidden'); }
    function enterDrawer(drawer) {
      closeDrawerModal();
      currentDrawer = drawer;
      document.getElementById('closet-drawer-select').value = drawer;
      switchTab('closet');
      renderCarousel();
    }
    function changeDrawerDirect(drawer) { currentDrawer = drawer; renderCarousel(); }
    let closetVault = JSON.parse(localStorage.getItem('toy_vault_v15') || '[]');
    let currentDrawer = 'All';
    let myWants = JSON.parse(localStorage.getItem('toy_wants_v4') || '[]');
    function sortLightToDark() {
      closetVault.sort((a, b) => (b.luminance || 0) - (a.luminance || 0));
    }
    function saveAndSync() {
      sortLightToDark();
      localStorage.setItem('toy_vault_v15', JSON.stringify(closetVault));
      document.getElementById('total-vault-pieces').innerText = closetVault.length + ' Pieces Stored';
      renderCarousel();
      renderWeeklyOutfits();
    }
