    function renderCarousel() {
      sortLightToDark();
      const items = (currentDrawer === 'All') ? closetVault : closetVault.filter(i => i.drawer === currentDrawer);
      const track = document.getElementById('carousel-track');
      if (items.length === 0) {
        track.innerHTML = '<div class="w-full glass rounded-3xl p-10 text-center space-y-2 border border-dashed border-slate-700"><i class="fa-solid fa-box-open text-4xl text-cyan-400 mb-1"></i><p class="text-sm font-bold text-white">' + currentDrawer + ' Drawer is Empty</p><p class="text-xs text-slate-400">Tap "+ Upload" to add items directly to this drawer.</p></div>';
        return;
      }
      track.innerHTML = items.map(item => `
        <div class="carousel-item glass rounded-3xl p-3 border border-slate-800 space-y-3 relative group" draggable="true" ondragstart="handleDragStart(event, '${item.id}')">
          <div class="w-full h-72 rounded-2xl bg-[#09111e] overflow-hidden relative cursor-zoom-in flex items-center justify-center p-2" onclick="openLightbox('${item.src}', '${item.drawer}')">
            <img src="${item.src}" class="max-h-full max-w-full object-contain drop-shadow-[0_12px_24px_rgba(0,0,0,0.85)]">
            <span class="absolute top-2 left-2 text-[10px] uppercase font-bold px-2.5 py-1 rounded-full bg-black/80 text-cyan-400 border border-cyan-400/40">${item.drawer}</span>
            <div class="absolute top-2 right-2 flex gap-1.5" onclick="event.stopPropagation()">
              <button onclick="togglePencilDrawer('${item.id}')" class="w-7 h-7 rounded-full bg-black/80 text-cyan-400 border border-cyan-400/50 flex items-center justify-center text-xs hover:bg-cyan-400 hover:text-black shadow"><i class="fa-solid fa-pencil"></i></button>
              <button onclick="toggleNotesBox('${item.id}')" class="w-7 h-7 rounded-full bg-black/80 text-purple-300 border border-purple-400/50 flex items-center justify-center text-xs hover:bg-purple-400 hover:text-black shadow"><i class="fa-solid fa-plus"></i></button>
            </div>
          </div>
          <div id="drawer-override-${item.id}" class="hidden p-2.5 bg-slate-950 border border-cyan-400 rounded-2xl space-y-1.5">
            <p class="text-[10px] text-cyan-400 font-bold uppercase">Move Piece To:</p>
            <div class="grid grid-cols-3 gap-1 text-[10px]">
              ${allDrawers.map(d => `<button onclick="reassignDrawer('${item.id}', '${d}')" class="p-1 rounded bg-slate-900 hover:bg-cyan-400 hover:text-black text-white truncate text-center ${item.drawer === d ? 'border border-cyan-400' : ''}">${d}</button>`).join('')}
            </div>
          </div>
          <div id="notes-box-${item.id}" class="${item.notes ? '' : 'hidden'} p-2.5 bg-slate-950/90 border border-purple-400/50 rounded-2xl space-y-1.5">
            <p class="text-[10px] text-purple-300 font-bold uppercase">Styling &amp; Piece Notes:</p>
            <textarea onchange="updateItemNotes('${item.id}', this.value)" placeholder="Add notes (e.g., Size, fabric, dry clean)..." class="w-full bg-slate-900 border border-slate-700 rounded-xl p-2 text-xs text-white h-14 resize-none">${item.notes || ''}</textarea>
          </div>
          <div class="flex items-center gap-2 pt-1">
            <button onclick="stageItemOnMannequin('${item.id}')" class="flex-1 py-2 bg-gradient-to-r from-teal-700 to-cyan-400 text-slate-950 font-bold rounded-xl text-xs active:scale-95 flex items-center justify-center gap-1.5">
              <i class="fa-solid fa-person-dress"></i> <span>Stage on Model</span>
            </button>
            <button onclick="deletePiece('${item.id}')" class="w-9 h-9 rounded-xl glass text-slate-500 hover:text-red-400 flex items-center justify-center"><i class="fa-solid fa-trash text-xs"></i></button>
          </div>
        </div>
      `).join('');
    }
