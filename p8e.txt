    function openLightbox(src, title) {
      document.getElementById('lightbox-img').src = src;
      document.getElementById('lightbox-title').innerText = title + ' (Full Resolution Cutout)';
      document.getElementById('image-lightbox').classList.remove('hidden');
    }
    function closeLightbox() { document.getElementById('image-lightbox').classList.add('hidden'); }
    function togglePencilDrawer(id) { document.getElementById('drawer-override-' + id).classList.toggle('hidden'); }
    function toggleNotesBox(id) { document.getElementById('notes-box-' + id).classList.toggle('hidden'); }
    function reassignDrawer(id, newDrawer) {
      const item = closetVault.find(i => String(i.id) === String(id));
      if (!item) return;
      item.drawer = newDrawer;
      saveAndSync();
    }
    function updateItemNotes(id, text) {
      const item = closetVault.find(i => String(i.id) === String(id));
      if (!item) return;
      item.notes = text;
      localStorage.setItem('toy_vault_v15', JSON.stringify(closetVault));
    }
    function deletePiece(id) {
      closetVault = closetVault.filter(i => String(i.id) !== String(id));
      saveAndSync();
    }
    let draggedItemId = null;
    function handleDragStart(e, id) { draggedItemId = id; e.dataTransfer.setData("text/plain", id); }
    function allowDrop(e) { e.preventDefault(); }
    function handleMiniDrop(e) {
      e.preventDefault();
      const id = e.dataTransfer.getData("text/plain") || draggedItemId;
      if (id) stageItemOnMannequin(id);
    }
    function stageItemOnMannequin(id) {
      const item = closetVault.find(i => String(i.id) === String(id));
      if (!item) return;
      const d = item.drawer.toLowerCase();
      let slotType = 'top';
      if (d === 'pants' || d === 'shorts' || d === 'skirts') slotType = 'bottom';
      else if (d === 'shoes') slotType = 'shoes';
      const mini = document.getElementById('mini-slot-' + slotType);
      mini.innerHTML = `<img src="${item.src}" class="w-full h-full object-contain">`;
      mini.style.borderColor = '#00F0FF';
      const stage = document.getElementById('stage-slot-' + slotType);
      stage.innerHTML = `<img src="${item.src}" class="w-full h-full object-contain"><span class="absolute bottom-1 right-2 text-[9px] bg-black/80 px-2 py-0.5 rounded text-cyan-400 font-bold">${item.drawer}</span>`;
      stage.style.borderColor = '#00F0FF';
    }
    let miniClickTimer = null;
    function onMiniModelClick() {
      if (miniClickTimer === null) {
        miniClickTimer = setTimeout(() => { miniClickTimer = null; openDressingRoom(); }, 260);
      } else {
        clearTimeout(miniClickTimer);
        miniClickTimer = null;
        resetDressedFit();
      }
    }
    function resetDressedFit() {
      document.getElementById('mini-slot-top').innerHTML = '<i class="fa-solid fa-shirt text-[9px] text-slate-500"></i>';
      document.getElementById('mini-slot-bottom').innerHTML = '<i class="fa-solid fa-person-dress text-[9px] text-slate-500"></i>';
      document.getElementById('mini-slot-shoes').innerHTML = '<i class="fa-solid fa-shoe-prints text-[8px] text-slate-500"></i>';
      document.getElementById('stage-slot-top').innerHTML = '<span class="text-xs text-slate-500"><i class="fa-solid fa-shirt mr-1"></i> Empty Top</span>';
      document.getElementById('stage-slot-bottom').innerHTML = '<span class="text-xs text-slate-500"><i class="fa-solid fa-person-dress mr-1"></i> Empty Bottom</span>';
      document.getElementById('stage-slot-shoes').innerHTML = '<span class="text-xs text-slate-500"><i class="fa-solid fa-shoe-prints mr-1"></i> Empty Shoes</span>';
    }
    function openDressingRoom() { document.getElementById('dressing-room-modal').classList.remove('hidden'); }
    function closeDressingRoom() { document.getElementById('dressing-room-modal').classList.add('hidden'); }
    function returnToDrawer(drawer) { closeDressingRoom(); enterDrawer(drawer); }
    function rotateModel(deg) {
      document.getElementById('mannequin-3d-stage').style.transform = `rotateY(${deg}deg)`;
      document.getElementById('rotation-angle-label').innerText = `${deg}°`;
    }
    let isAutoSpinning = false, spinInterval = null;
    function toggleAutoSpin() {
      const btn = document.getElementById('auto-spin-btn'), slider = document.getElementById('rotation-slider');
      if (!isAutoSpinning) {
        isAutoSpinning = true;
        btn.innerText = 'Auto Spin: ON';
        btn.classList.replace('bg-cyan-400', 'bg-purple-500');
        spinInterval = setInterval(() => {
          let cur = (parseInt(slider.value) + 2) % 360;
          slider.value = cur;
          rotateModel(cur);
        }, 40);
      } else {
        isAutoSpinning = false;
        clearInterval(spinInterval);
        btn.innerText = 'Auto Spin: OFF';
        btn.classList.replace('bg-purple-500', 'bg-cyan-400');
      }
    }
